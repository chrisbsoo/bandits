\subsection{Model (Prototype)}
$$\text{The model depends on hyperparameters }$$
$$\sigma_1, \ldots ,\sigma_K, \lambda_1^{(0)}, \ldots, \lambda_K^{(0)}, \lambda_1^{(1)}, \ldots, \lambda_K^{(1)}, n_1, \ldots, n_K, c_1, \ldots, c_K$$
$$1, 2, \ldots, K\; \text{arms}, \; 1, 2, \ldots, T\; \text{Time Horizon}$$
$$Y^{(\sigma_i)}_{i, t} \sim N(Z_{i, t}, \sigma_i^2)\; \text{Reward of arm $k$ at time $t$}$$
$$Z_{i, t} = \mathbb{E}(Y_{i, t}^{(\sigma)}) \in \mathcal{Z}$$
$$\mathcal{Z} = \left\{ Z \in \{0, 1\}^{K \times T} : \sum_{t=1}^T Z_{i, t} = n_i, \; \sum_{t=1}^{T-1}Z_{i, t} Z_{i, t+1} \geq c_i \right\}$$
$$a_t \subseteq \{1, \ldots, K\} \text{ active arms at time }t$$
$$\text{Let }1_{i, t} = 1(i \in a_t), \; \text{Let }H_i^{(0)}(t) = \min\{s \geq 1: 1_{i, t+s} \neq 1_{i,t}, \; 1_{i,t} = 0\} \text{ Sleeping holding time}$$
$$\text{Let }H_i^{(1)}(t) = \min\{s \geq 1: 1_{i, t+s} \neq 1_{i, t}, \; 1_{i,t} = 1\} \text{ Awake holding time}$$
$$H_i^{(j)}(t) \sim \text{Geom}(\lambda_i^{(j)}), \;i.e\; \mathbb{P}(H_i^{(j)}(t) = s) = (1-\lambda^{(j)}_i)^{s-1}\lambda^{(j)}_i, \;\; s \geq 1$$
$$\text{Let }X_t \text{ be the decisions of the algorithm $f \in \mathcal{F}_t$ } s.t \; \mathcal{F}_t = \left\{f : \{Y_{X_s, s}^{(\sigma_{X_s})}, X_{s}\}_{s=1}^t \rightarrow \{1, \ldots, K\}\right\} \; \forall \; t\in\{1, \ldots, T\}$$
$$\text{We now formalize the regret:}$$
$$R_T = \mathbb{E}\left(\sum_{t=1}^T Y^{(\sigma_{i^*})}_{i^*, t} 1_{i^*, t} - \sum_{t=1}^T Y_{X_t, t}^{(\sigma_{X_t})}\right) = \sum_{t=1}^T Z_{i^*, t} 1_{i^*, t} - \sum_{t=1}^T Z_{X_t, t}$$
$$i^* = \arg\max_{i \in [K]} \sum_{t=1}^T Z_{i, t}1_{i, t}$$

\section{A1}

\begin{algorithm}
    \caption{Generate Adversarial Sequence}
    \begin{algorithmic}[1]
        \Require Horizon $T$, Autocorrelation Parameter $c$, Observation Ratio $\rho$
        \Ensure Sequence $X_1, X_2, \ldots , X_T$ where $X_i \in \{0, 1\}, \;\frac{1}{T}\sum_{t=1}^T X_t = \rho$
        \State $X_1 \sim Bernoulli(\rho)$
        \For{$t = 2, \dots, T$}
            \State $\hat{p}_t \gets \frac{1}{t-1} \sum_{s=1}^{t-1} X_s$ \Comment{Running empirical mean}
            \State $e_t \gets \hat{p}_t - \rho$ \Comment{Error from target}

            \If{$X_{t-1} = 1$}
                \State $p_{\text{stay}} \gets \frac{1+c}{2} - e_t$
            \Else
                \State $p_{\text{stay}} \gets \frac{1+c}{2} + e_t$
            \EndIf

            \State $p_{\text{stay}} \gets \text{clip}(p_{\text{stay}},\ 0,\ 1)$
            \State $\text{stay} \sim \text{Bernoulli}(p_{\text{stay}})$

            \If{$\text{stay}$}
                \State $X_t \gets X_{t-1}$
            \Else
                \State $X_t \gets 1 - X_{t-1}$
            \EndIf
        \EndFor

        \State \Return $X_1, \dots, X_T$

    \end{algorithmic}
\end{algorithm}
