\section{Mathematical Framework}

\subsection{Model (Prototype)}
$$\text{The model depends on hyperparameters }$$
$$\sigma_1, \ldots ,\sigma_K, \lambda_1^{(0)}, \ldots, \lambda_K^{(0)}, \lambda_1^{(1)}, \ldots, \lambda_K^{(1)}, n_1, \ldots, n_K, c_1, \ldots, c_K$$
$$1, 2, \ldots, K\; \text{arms}, \; 1, 2, \ldots, T\; \text{Time Horizon}$$
$$Y^{(\sigma_i)}_{i, t} \sim N(Z_{i, t}, \sigma_i^2)\; \text{Reward of arm $k$ at time $t$}$$
$$Z_{i, t} = \mathbb{E}(Y_{i, t}^{(\sigma)}) \in \mathcal{Z}$$
$$\mathcal{Z} = \left\{ Z \in \{0, 1\}^{K \times T} : \sum_{t=1}^T Z_{i, t} = n_i, \; \frac{1}{T}\sum_{t=1}^{T-1}Z_{i, t} Z_{i, t+1} \geq c_i \right\}$$
$$a_t \subseteq \{1, \ldots, K\} \text{ active arms at time }t$$
$$\text{Let }1_{i, t} = 1(i \in a_t), \; \text{Let }H_i^{(0)}(t) = \min\{s \geq 1: 1_{i, t+s} \neq 1_{i,t}, \; 1_{i,t} = 0\} \text{ Sleeping holding time}$$
$$\text{Let }H_i^{(1)}(t) = \min\{s \geq 1: 1_{i, t+s} \neq 1_{i, t}, \; 1_{i,t} = 1\} \text{ Awake holding time}$$
$$H_i^{(j)}(t) \sim \text{Geom}(\lambda_i^{(j)}), \;i.e\; \mathbb{P}(H_i^{(j)}(t) = s) = (1-\lambda^{(j)}_i)^{s-1}\lambda^{(j)}_i, \;\; s \geq 1$$
$$\text{Let }X_t \text{ be the decisions of the algorithm $f \in \mathcal{F}_t$ } s.t \; \mathcal{F}_t = \left\{f : \{Y_{X_s, s}^{(\sigma_{X_s})}, X_{s}\}_{s=1}^t \rightarrow a_t \right\} \; \forall \; t\in\{1, \ldots, T\}$$
$$\text{We now formalize the regret:}$$
$$R_T = \mathbb{E}\left(\sum_{t=1}^T Y^{(\sigma_{i^*})}_{i^*, t} 1_{i^*, t} - \sum_{t=1}^T Y_{X_t, t}^{(\sigma_{X_t})}\right) = \sum_{t=1}^T Z_{i^*, t} 1_{i^*, t} - \sum_{t=1}^T Z_{X_t, t}$$
$$i^* = \arg\max_{i \in [K]} \sum_{t=1}^T Z_{i, t}1_{i, t}$$

\subsection{Generating A (two state markov chain)}

$$A_{i, t} = \{1, 0\}^{K \times T}$$
$$\text{Initialise }A_{i, 1} = 1 \;\forall \;i$$
$$\text{For }t = 2, 3, \ldots, T$$
$$U_{i, t} \sim Unif(0, 1)$$
$$A_{i, t} = 1(A_{i, t-1} = 1 \cap U_{i, t} \geq \lambda_i^{(1)}) + 1(A_{i, t-1} = 0\cap U < \lambda_i^{(0)})$$

\subsection{Generating Z (partitions + stars and bars selection)}

$$M \sim Unif(1, \ldots , M_{max}), \; M_{max} = \min(n_i-c_i T, T - n_i + 1)$$
$$\text{Choose $M-1$ dividers uniformly without replacement from $\{1, \ldots, n_i - 1\}$}$$:
$$d_1 < d_2 < \cdots < d_{M-1}, \quad d_k \sim \text{Unif}\binom{[n_i-1]}{M-1}$$

$$\text{Setting $d_0 = 0$ and $d_M = n_i$, the cluster sizes are:}$$
$$s_j = d_j - d_{j-1}, \quad j = 1, \ldots, M$$

$$\text{This samples uniformly over all compositions of $n_i$ into $M$ positive parts:}$$
$$s_1 + s_2 + \cdots + s_M = n_i, \quad s_j \geq 1$$

$$\text{with each of the $\binom{n_i - 1}{M-1}$ compositions equally likely.}$$

\subsection{Generating Y}

$$Y_{i, t}^{(\sigma_i)} \sim N(Z_{i, t}, \sigma_i) $$

$$\text{and passing a NaN through a gaussian rng doesn't change the value.}$$
